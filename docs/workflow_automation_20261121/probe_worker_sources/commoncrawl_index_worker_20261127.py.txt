import sys,json,urllib.request,urllib.error,base64,hashlib
url=sys.argv[1]
if not url.startswith('https://index.commoncrawl.org/CC-MAIN-2024-10-index?url='):raise ValueError('FROZEN_INDEX_ONLY')
class NoRedirect(urllib.request.HTTPRedirectHandler):
 def redirect_request(self,*args,**kwargs):raise urllib.error.HTTPError(url,302,'REDIRECT_REJECTED',{},None)
req=urllib.request.Request(url,headers={'User-Agent':'4511HistoricalSourceProbe/0.1','Accept':'application/json'})
print(json.dumps({'event':'HTTP_INTENT','url':url}),flush=True)
try:
 with urllib.request.build_opener(NoRedirect()).open(req,timeout=8) as response:
  raw=response.read(65537)
  value=dict(event='HTTP_RESULT',status=response.status,final_url=response.url,body_bytes=len(raw),
   body_sha256=hashlib.sha256(raw).hexdigest(),body_b64=base64.b64encode(raw).decode(),too_large=len(raw)>65536)
except urllib.error.HTTPError as exc:
 value=dict(event='HTTP_RESULT',status=exc.code,error_type=type(exc).__name__)
except Exception as exc:value=dict(event='HTTP_FAILED',error_type=type(exc).__name__)
print(json.dumps(value),flush=True)
